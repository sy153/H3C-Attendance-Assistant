    maximum,
    fallback

      return fallback;

    return Math.min(
      maximum,
      Math.max(minimum, number)
    );

  function parseDate(value) {
    const match =
      String(value || "").match(
        /(\d{4})[年/-](\d{1,2})[月/-](\d{1,2})/
      );

    if (!match) {
      return null;
    }

    const year = Number(match[1]);
    const month = Number(match[2]);
    const day = Number(match[3]);

    const date =
      new Date(
        year,
        month - 1,
        day
      );

    if (
      date.getFullYear() !== year ||
      date.getMonth() !== month - 1 ||
      date.getDate() !== day
    ) {
      return null;
    }

    date.setHours(0, 0, 0, 0);

    return date;
  }

  function getCellDate(cell) {
    const title =
      cell?.getAttribute?.("title") || "";

    return parseDate(
      title.split("｜")[0].trim()
    );
  }

  function timeToMinutes(value) {
    const match =
      String(value || "").match(
        /^(\d{1,2}):(\d{2})$/
      );

    if (!match) {
      return null;
    }

    const hour = Number(match[1]);
    const minute = Number(match[2]);

    if (
      hour < 0 ||
      hour > 23 ||
      minute < 0 ||
      minute > 59
    ) {
      return null;
    }

    return hour * 60 + minute;
  }

  function formatClock(minutes) {
    if (!Number.isFinite(minutes)) {
      return "--:--";
    }

    const dayMinutes = 24 * 60;
    const nextDay = minutes >= dayMinutes;

    const normalized =
      (
        minutes % dayMinutes +
        dayMinutes
      ) % dayMinutes;

    const text =
      `${pad2(
        Math.floor(normalized / 60)
      )}:${pad2(normalized % 60)}`;

    return nextDay
      ? `${text} 次日`
      : text;
  }

  function formatHours(
    minutes,
    showPositiveSign = true
  ) {
    if (!Number.isFinite(minutes)) {
      return "--";
    }

    const normalized =
      Math.abs(minutes) < 0.5
        ? 0
        : minutes;

    const sign =
      showPositiveSign &&
      normalized > 0
        ? "+"
        : "";

    return (
      `${sign}` +
      `${(normalized / 60).toFixed(2)} 小时`
    );
  }

  function formatCompact(minutes) {
    if (!Number.isFinite(minutes)) {
      return "--";
    }

    const normalized =
      Math.abs(minutes) < 0.5
        ? 0
        : minutes;

    const sign =
      normalized > 0
        ? "+"
        : "";

    return (
      `${sign}` +
      `${(normalized / 60).toFixed(2)}h`
    );
  }

  function formatShort(minutes) {
    if (!Number.isFinite(minutes)) {
      return "--";
    }

    const rounded =
      Math.round(minutes);

    const sign =
      rounded < 0
        ? "-"
        : "";

    const absolute =
      Math.abs(rounded);

    const hours =
      Math.floor(absolute / 60);

    const remain =
      absolute % 60;

    if (!hours) {
      return `${sign}${remain}分钟`;
    }

    if (!remain) {
      return `${sign}${hours}小时`;
    }

    return (
      `${sign}${hours}小时` +
      `${remain}分钟`
    );
  }

  function overlap(
    start1,
    end1,
    start2,
    end2
  ) {
    if (
      ![
        start1,
        end1,
        start2,
        end2
      ].every(Number.isFinite)
    ) {
      return 0;
    }

    return Math.max(
      0,
      Math.min(end1, end2) -
        Math.max(start1, start2)
    );
  }

  function standardMinutes(rules) {
    return (
      rules.normalEnd -

        rules.normalEnd,

      )
    );
  }

    );
  }

  function detectMonth(cells) {
    const currentCells =
      cells.filter(
        isCurrentMonthCell
      );

    const source =
      currentCells.length
        ? currentCells
        : cells;

    const counts =
      new Map();

    for (const cell of source) {
      const date =
        getCellDate(cell);

      if (!date) {
        continue;
      }

      const key =
        getMonthKey(
          date.getFullYear(),
          date.getMonth() + 1
        );

      counts.set(
        key,
        (counts.get(key) || 0) + 1
      );
    }

    const best =
      [...counts.entries()]
        .sort(
          (a, b) =>
            b[1] - a[1]
        )[0];

    if (!best) {
      return null;
    }

    const [year, month] =
      best[0]
        .split("-")
        .map(Number);

    return {
      year,
      month
    };
  }

  function hasClassFragment(
    element,
    fragment
  ) {
    if (!element) {
      return false;
    }

    // 性能：递归检查自身及子孙节点的 class 字符串，
    // 避免每个格子都触发 CSS 选择器引擎开销
    if (
      String(
        element.className?.baseVal ??
          (element.className || "")
      ).includes(fragment)
    ) {
      return true;
    }

    for (const child of element.children || []) {
      if (hasClassFragment(child, fragment)) {
        return true;
      }
    }

    return false;
  }

  function getDayInfo(date, cell) {

    // 合并文本与 title 一起检查
    const combined = `${text} ${title}`;

    // 1. 明确的调休上班标记
    if (
      /调休上班|补班|正常上班/.test(combined) ||
      hasClassFragment(cell, "holiday1") ||
      hasClassFragment(cell, "holiday_work") ||
      hasClassFragment(cell, "holiday-work")
    ) {
      return {
        type: "workday",
        label: "工作日",
        mode: "workday",
        isWeekend: false,
        isSpecial: false
      };
    }

    // 2. 特殊假期/出差等
    for (const rule of SPECIAL_DAY_RULES) {
      if (rule.pattern.test(combined)) {
        return {
          type: rule.type,
          label: rule.label,
          mode: rule.mode,
          isWeekend: false,
          isSpecial: true
        };
      }
    }

    // 3. 周末调休兜底：日期是周末，但含“上班”且不含“休息/假期”，视为工作日
    if (isNaturalWeekend(date)) {
      if (/上班/.test(combined) && !/休息|假期/.test(combined)) {
        return {
          type: "workday",
          label: "工作日",
          mode: "workday",
          isWeekend: false,
          isSpecial: false
        };
      }

      // 有 holiday 类则视为休息日
      if (hasClassFragment(cell, "holiday")) {
        return {
          type: "weekend",
          label: "休息日",
          mode: "weekend",
          isWeekend: true,
          isSpecial: false
        };
      }

      // 否则就是普通休息日
      return {
        type: "weekend",
        label: "休息日",
        mode: "weekend",
        isWeekend: true,
        isSpecial: false
      };
    }

    // 4. 默认普通工作日
    return {
      type: "workday",
      label: "工作日",
      mode: "workday",
      isWeekend: false,
      isSpecial: false
    };
  }

  function parseTimeToken(value) {
    const match =
      String(value || "").match(
        /(\d{1,2}):(\d{2})/
      );

    if (!match) {
      return null;
    }

    const hour = Number(match[1]);
    const minute = Number(match[2]);

    if (
      hour < 0 ||
      hour > 23 ||
      minute < 0 ||
      minute > 59
    ) {
      return null;
    }

    return hour * 60 + minute;
  }

  function parsePunchText(text) {
    const source =
      String(text || "")
        .replace(/\s+/g, " ")
        .trim();

    if (!source) {
      return null;
    }

    const range =
      source.match(
        /(\d{1,2}):(\d{2})\s*(?:-|~|～|至)\s*(\d{1,2}):(\d{2})/
      );

    if (range) {
      const clockIn =
        Number(range[1]) * 60 +
        Number(range[2]);

      let clockOut =
        Number(range[3]) * 60 +
        Number(range[4]);

      if (
        source.includes("次日") ||
        clockOut < clockIn
      ) {
        clockOut += 24 * 60;
      }

      return {
        clockIn,
        clockOut,
        hasMorning: true,
        hasAfternoon: true
      };
    }

    let clockIn = null;
    let clockOut = null;

    const morning =
      source.match(
        /(?:上午|上班|签到|上班卡|上班打卡)[^\d]{0,15}(\d{1,2}:\d{2})/
      );

    const afternoon =
      source.match(
        /(?:下午|下班|签退|下班卡|下班打卡)[^\d]{0,15}(\d{1,2}:\d{2})/
      );

    if (morning) {
      clockIn =
        parseTimeToken(
          morning[1]
        );
    }

    if (afternoon) {
      clockOut =
        parseTimeToken(
          afternoon[1]
        );
    }

    if (
      clockIn === null &&
      clockOut === null
    ) {
      const times =
        (
          source.match(
            /\b\d{1,2}:\d{2}\b/g
          ) || []
        )
          .map(parseTimeToken)
          .filter(Number.isFinite);

      if (times.length >= 2) {
        clockIn = times[0];

        clockOut =
          times[times.length - 1];
      } else if (times.length === 1) {
        const onlyTime =
          times[0];

        if (
          /下午|下班|签退/.test(source)
        ) {
          clockOut = onlyTime;
        } else if (
          /上午|上班|签到/.test(source)
        ) {
          clockIn = onlyTime;
        } else if (
          onlyTime < 12 * 60
        ) {
          clockIn = onlyTime;
        } else {
          clockOut = onlyTime;
        }
      }
    }

    if (
      clockIn === null &&
      clockOut === null
    ) {
      return null;
    }

    if (
      clockIn !== null &&
      clockOut !== null &&
      (
        source.includes("次日") ||
        clockOut < clockIn
      )
    ) {
      clockOut += 24 * 60;
    }

    return {
      clockIn,
      clockOut,

      hasMorning:
        clockIn !== null,

      hasAfternoon:
        clockOut !== null
    };
  }

  function getPunch(cell) {
    for (
      const selector
      of CONFIG.timeSelectors
    ) {
      const element =
        cell.querySelector(selector);

      if (!element) {
        continue;
      }

      const result =
        parsePunchText(
          element.textContent
        );

      if (result) {
        return result;
      }
    }

    return parsePunchText(
      cell.textContent
    );
  }

  function parseLeaveTime(cell) {
    const text =
      String(
        cell.textContent || ""
      )
        .replace(/\s+/g, " ")
        .trim();

    const match =
      text.match(
        /[（(](\d{1,2}:\d{2})\s*[-~至]\s*(\d{1,2}:\d{2})[)）]/
      );

    if (!match) {
      return null;
    }

    const start = timeToMinutes(match[1]);
    const end = timeToMinutes(match[2]);

    if (start === null || end === null) {
      return null;
    }

    const leaveStart = start;
    const leaveEnd = end < start ? end + 24 * 60 : end;

    return { start: leaveStart, end: leaveEnd };
  }

  function cellFingerprint(
    cell,
    dateKey
  ) {
    const text =
      String(
        cell.textContent || ""
      )
        .replace(/\s+/g, " ")
        .trim();

    return [
      dateKey,
      cell.className || "",
      text
    ].join("|");
  }

  function parseCell(
    cell,
    date,
    dateKey
  ) {
    const fingerprint =
      cellFingerprint(
        cell,
        dateKey
      );

    const cached =
      state.cellCache.get(dateKey);

    if (
      cached?.fingerprint ===
      fingerprint
    ) {
      return cached.record
        ? {
            ...cached.record,
            date: new Date(date)
          }
        : null;
    }

    const dayInfo =
      getDayInfo(date, cell);

    const punch =
      getPunch(cell);

    let record = null;

    if (
      dayInfo.isSpecial ||
      punch
    ) {
      record = {
        date: new Date(date),
        dateKey,

        dayType: dayInfo.type,
        dayLabel: dayInfo.label,
        dayMode: dayInfo.mode,

        isSpecial:
          dayInfo.isSpecial,

        isWeekend:
          dayInfo.isWeekend,

        clockIn:
          punch?.clockIn ?? null,

        clockOut:
          punch?.clockOut ?? null,

        hasMorning:
          punch?.hasMorning ?? false,

        hasAfternoon:
          punch?.hasAfternoon ?? false,

        leaveStart: null,
        leaveEnd: null
      };

      if (
        record.isSpecial &&
        record.dayMode === "leave"
      ) {
        const leaveTime =
          parseLeaveTime(cell);

        if (leaveTime) {
          record.leaveStart =
            leaveTime.start;

          record.leaveEnd =
            leaveTime.end;
        }
      }
    }

    state.cellCache.set(
      dateKey,
      {
        fingerprint,

        record:
          record
            ? {
                ...record,
                date: null
              }
            : null
      }
    );

    return record;
  }

  function extractRecords() {
    const cells =
      getCells();

    const displayedMonth =
      detectMonth(cells);

    if (!displayedMonth) {
      return {
        displayedMonth: null,
        records: [],
        cells,
        cellsByDate: new Map()
      };
    }

    const monthPrefix =
      getMonthKey(
        displayedMonth.year,
        displayedMonth.month
      );

    for (
      const key
      of state.cellCache.keys()
    ) {
      if (
        !key.startsWith(
          monthPrefix
        )
      ) {
        state.cellCache.delete(key);
      }
    }

    const records =
      new Map();

    const cellsByDate =
      new Map();

    for (const cell of cells) {
      const date =
        getCellDate(cell);

      if (
        !date ||
        date.getFullYear() !==
          displayedMonth.year ||
        date.getMonth() + 1 !==
          displayedMonth.month
      ) {
        continue;
      }

      const dateKey =
        getDateKey(date);

      cellsByDate.set(
        dateKey,
        cell
      );

      if (
        CONFIG.excludeToday &&
        isToday(date)
      ) {
        continue;
      }

      const record =
        parseCell(
          cell,
          date,
          dateKey
        );

      if (!record) {
        continue;
      }

      const old =
        records.get(dateKey);

      const oldScore =
        old
          ? (
            Number(old.hasMorning) +
            Number(old.hasAfternoon) +
            Number(old.isSpecial) * 2
          )
          : -1;

      const newScore =
        Number(record.hasMorning) +
        Number(record.hasAfternoon) +
        Number(record.isSpecial) * 2;

      if (
        !old ||
        newScore > oldScore
      ) {
        records.set(
          dateKey,
          record
        );
      }
    }

    return {
      displayedMonth,

      records:
        [...records.values()]
          .sort(
            (a, b) =>
              a.date - b.date
          ),

      cells,
      cellsByDate
    };
  }

  function getInvalidReasons(
    record,
    rules
  ) {
    if (record.isSpecial && !record.leaveStart) {
      return [];
    }

    const reasons = [];

    if (
      record.clockIn === null &&
      record.clockOut !== null
    ) {
      reasons.push(
        "缺少上班时间"
      );
    }

    if (
      record.clockIn !== null &&
      record.clockOut === null
    ) {
      reasons.push(
        "缺少下班时间"
      );
    }

    if (
      !record.isWeekend &&
      Number.isFinite(
        record.clockIn
      ) &&
      Number.isFinite(
        record.clockOut
      )
    ) {
      if (
        record.clockIn >
        rules.flexibleLatest
      ) {
        reasons.push(
          `上班晚于弹性最晚时间 ` +
          formatClock(
            rules.flexibleLatest
          )
        );
      }

      if (
        record.clockOut <
        rules.normalEnd
      ) {
        reasons.push(
          `下班早于正常下班时间 ` +
          formatClock(
            rules.normalEnd
          )
        );
      }
    }

    return reasons;
  }

  function calculateRecord(
    record,
    rules,
    standard
  ) {
    if (record.isSpecial && record.leaveStart != null) {
      const leaveSpan =
        record.leaveEnd -
        record.leaveStart;

      const leaveLunch =
        overlap(
          record.leaveStart,
          record.leaveEnd,
          rules.lunchStart,
          rules.lunchEnd
        );

      const leaveNet =
        Math.max(
          0,
          leaveSpan - leaveLunch
        );

      const expectedNet =
        standard - leaveNet;

      if (
        !Number.isFinite(
          record.clockIn
        ) ||
        !Number.isFinite(
          record.clockOut
        )
      ) {
        return {
          overtimeMinutes:
            -expectedNet,

          actualNetMinutes: null,
          lunchMinutes: null,
          expectedClockOut: null,

          isInvalid: true,

          invalidReasons: [
            "打卡时间不完整"
          ]
        };
      }

      const span =
        Math.max(
          0,
          record.clockOut -
            record.clockIn
        );

      if (span > 18 * 60) {
        return {
          overtimeMinutes: 0,
          actualNetMinutes: null,
          lunchMinutes: null,
          expectedClockOut: null,

          isInvalid: true,

          invalidReasons: [
            "打卡时间跨度超过18小时"
          ]
        };
      }

      const lunchMinutes =
        overlap(
          record.clockIn,
          record.clockOut,
          rules.lunchStart,
          rules.lunchEnd
        );

      const actualNetMinutes =
        Math.max(
          0,
          span - lunchMinutes
        );

      const overtime =
        actualNetMinutes -
        expectedNet;

      return {
        overtimeMinutes: overtime,
        actualNetMinutes,
        lunchMinutes,
        expectedClockOut: null,

        isInvalid: false,
        invalidReasons: []
      };
    }

    if (record.isSpecial) {
      return {
        overtimeMinutes: 0,
        actualNetMinutes: standard,
        lunchMinutes: 0,
        expectedClockOut: null,

        isInvalid: false,
        invalidReasons: []
      };
    }

    const invalidReasons =
      getInvalidReasons(
        record,
        rules
      );

    if (invalidReasons.length) {
      return {
        overtimeMinutes: 0,
        actualNetMinutes: null,
        lunchMinutes: null,

        expectedClockOut:
          record.isWeekend
            ? null
            : expectedClockOut(
                record.clockIn,
                rules
              ),

        isInvalid: true,
        invalidReasons
      };
    }

    if (
      !Number.isFinite(
        record.clockIn
      ) ||
      !Number.isFinite(
        record.clockOut
      )
    ) {
      return {
        overtimeMinutes: 0,
        actualNetMinutes: null,
        lunchMinutes: null,
        expectedClockOut: null,

        isInvalid: true,

        invalidReasons: [
          "打卡时间不完整"
        ]
      };
    }

    const span =
      Math.max(
        0,
        record.clockOut -
          record.clockIn
      );

    if (span > 18 * 60) {
      return {
        overtimeMinutes: 0,
        actualNetMinutes: null,
        lunchMinutes: null,
        expectedClockOut: null,

        isInvalid: true,

        invalidReasons: [
          "打卡时间跨度超过18小时"
        ]
      };
    }

    const lunchMinutes =
      overlap(
        record.clockIn,
        record.clockOut,
        rules.lunchStart,
        rules.lunchEnd
      );

    const actualNetMinutes =
      Math.max(
        0,
        span - lunchMinutes
      );

    return {
      overtimeMinutes:
        record.isWeekend
          ? actualNetMinutes
          : actualNetMinutes -
            standard,

      actualNetMinutes,
      lunchMinutes,

      expectedClockOut:
        record.isWeekend
          ? null
          : expectedClockOut(
              record.clockIn,
              rules
            ),

      isInvalid: false,
      invalidReasons: []
    };
  }

  function summarize(records) {
    const rules =
      getRules();

    const standard =
      standardMinutes(rules);

    const summary = {
      totalMinutes: 0,
      weekdayMinutes: 0,
      weekendMinutes: 0,

      weekdayDays: 0,
      weekendDays: 0,

      details: []
    };

    for (const record of records) {
      const detail = {
        ...record,

        ...calculateRecord(
          record,
          rules,
          standard
        )
      };

      summary.details.push(detail);

      if (record.dayMode === "official") {
        summary.weekdayDays += 1;
      }
      else if (
        record.isSpecial &&
        !record.leaveStart
      ) {
        // 跳过
      }
      else if (
        record.isSpecial &&
        record.leaveStart != null
      ) {
        summary.weekdayDays += 1;
      }
      else {
        if (record.isWeekend) {
          summary.weekendDays += 1;
        } else {
          summary.weekdayDays += 1;
        }
      }

      if (
        detail.isInvalid ||
        (
          record.isSpecial &&
          !record.leaveStart
        )
      ) {
        continue;
      }

      if (record.isWeekend) {
        summary.weekendMinutes +=
          detail.overtimeMinutes;
      } else {
        summary.weekdayMinutes +=
          detail.overtimeMinutes;

        summary.totalMinutes +=
          detail.overtimeMinutes;
      }
    }

    return summary;
  }

  function remainingWorkdays(
    displayedMonth,
    cells
  ) {
    const today =
      new Date();

    today.setHours(0, 0, 0, 0);

    const monthStart =
      new Date(
        displayedMonth.year,
        displayedMonth.month - 1,
        1
      );

    const monthEnd =
      new Date(
        displayedMonth.year,
        displayedMonth.month,
        0
      );

    monthStart.setHours(0, 0, 0, 0);
    monthEnd.setHours(0, 0, 0, 0);

    const current =
      displayedMonth.year ===
        today.getFullYear() &&
      displayedMonth.month ===
        today.getMonth() + 1;

    const future =
      monthStart > today;

    if (monthEnd <= today) {
      return {
        count: 0,
        current: false,
        future: false
      };
    }

    const typeMap =
      new Map();

    for (const cell of cells) {
      const date =
        getCellDate(cell);

      if (!date) {
        continue;
      }

      typeMap.set(
        getDateKey(date),
        getDayInfo(date, cell)
      );
    }

    const first =
      current
        ? new Date(
            today.getFullYear(),
            today.getMonth(),
            today.getDate() + 1
          )
        : monthStart;

    let count = 0;

    for (
      const date = new Date(first);
      date <= monthEnd;
      date.setDate(
        date.getDate() + 1
      )
    ) {
      const copy =
        new Date(date);

      const info =
        typeMap.get(
          getDateKey(copy)
        );

      if (info) {
        if (
          info.mode === "workday" ||
          info.mode === "official"
        ) {
          count += 1;
        }

        continue;
      }

      if (!isNaturalWeekend(copy)) {
        count += 1;
      }
    }

    return {
      count,
      current,
      future
    };
  }

  function saveHistory(
    month,
    summary,
    recordCount
  ) {
    const key =
      getMonthKey(
        month.year,
        month.month
      );

    const old =
      state.history[key];

    if (
      old?.recordCount > 0 &&
      recordCount === 0
    ) {
      return;
    }

    const next = {
      year: month.year,
      month: month.month,

      totalMinutes:
        summary.totalMinutes,

      weekdayMinutes:
        summary.weekdayMinutes,

      weekendMinutes:
        summary.weekendMinutes,

      weekdayDays:
        summary.weekdayDays,

      weekendDays:
        summary.weekendDays,

      recordCount,
      updatedAt: Date.now()
    };

    const oldComparable =
      old
        ? {
            ...old,
            updatedAt: 0
          }
        : null;

    const nextComparable = {
      ...next,
      updatedAt: 0
    };

    if (
      oldComparable &&
      JSON.stringify(
        oldComparable
      ) ===
        JSON.stringify(
          nextComparable
        )
    ) {
      return;
    }

    state.history[key] =
      next;

    const keys =
      Object.keys(
        state.history
      ).sort();

    while (keys.length > 36) {
      delete state.history[
        keys.shift()
      ];
    }

    writeStorage(
      CONFIG.historyKey,
      state.history
    );
  }

  function stageHistorySave(result) {
    clearTimeout(
      state.stableTimer
    );

    const token =
      recordsToken(
        result.displayedMonth,
        result.records
      );

    state.pendingHistoryToken =
      token;

    state.stableTimer =
      window.setTimeout(
        () => {
          state.stableTimer = null;

          try {
            const fresh =
              extractRecords();

            if (
              !fresh.displayedMonth ||
              !state.pendingHistoryToken
            ) {
              return;
            }

            const freshToken =
              recordsToken(
                fresh.displayedMonth,
                fresh.records
              );

            if (
              freshToken !==
              state.pendingHistoryToken
            ) {
              return;
            }

            saveHistory(
              fresh.displayedMonth,
              summarize(
                fresh.records
              ),
              fresh.records.length
            );
          } catch (error) {
            console.debug(
              "[H3C 考勤助手] 保存趋势失败：",
              error
            );
          }
        },
        CONFIG.historyStableDelay
      );
  }

  function getTrendData(
    displayedMonth,
    count
  ) {
    const base =
      new Date(
        displayedMonth.year,
        displayedMonth.month - 1,
        1
      );

    const result = [];

    for (
      let offset = count - 1;
      offset >= 0;
      offset -= 1
    ) {
      const date =
        new Date(
          base.getFullYear(),
          base.getMonth() - offset,
          1
        );

      const year =
        date.getFullYear();

      const month =
        date.getMonth() + 1;

      const key =
        getMonthKey(
          year,
          month
        );

      const historyData =
        state.history[key] || null;

      result.push({
        key,
        year,
        month,

        data:
          historyData
            ? {
                ...historyData,

                totalMinutes:
                  Number.isFinite(
                    historyData.weekdayMinutes
                  )
                    ? historyData.weekdayMinutes
                    : historyData.totalMinutes
              }
            : null
      });
    }

    return result;
  }

  function renderTrend(

        };
      }
    }

    const values =

        )

    const maximum =
      Math.max(
        60,

      );

    const width = 420;
    const height = 180;
    const baseline = 80;
    const barMax = 55;
    const padding = 12;

    const slot =
      (
        width -
        padding * 2
      ) / count;

    const barWidth =
      count === 12
        ? 16
        : 27;

    const svg = [
      `
        <line
          x1="${padding}"
          y1="${baseline}"
          x2="${width - padding}"
          y2="${baseline}"
          class="h3c-chart-axis"
        ></line>
      `
    ];

    months.forEach(
      (item, index) => {
        const x =
          padding +
          slot * index +
          slot / 2;

        const value =
          item.data?.totalMinutes;

        if (Number.isFinite(value)) {
          const barHeight =
            Math.max(
              2,
              Math.abs(value) /
                maximum *
                barMax
            );

          const y =
            value >= 0
              ? baseline - barHeight
              : baseline;

          const color =
            value > 0
              ? "#10b981"
              : value < 0
                ? "#f43f5e"
                : "#60a5fa";

          const textY =
            value >= 0
              ? y - 6
              : baseline +
                barHeight +
                12;

          svg.push(`
            <rect
              x="${x - barWidth / 2}"
              y="${y}"
              width="${barWidth}"
              height="${barHeight}"
              rx="4"
              fill="${color}"
              fill-opacity=".84"
            ></rect>

            <text
              x="${x}"
              y="${textY}"
              text-anchor="middle"
              class="h3c-chart-value"
              fill="${color}"
            >
              ${(value / 60).toFixed(1)}
            </text>
          `);
        } else {
          svg.push(`
            <circle
              cx="${x}"
              cy="${baseline}"
              r="2.5"
              class="h3c-chart-empty"
            ></circle>
          `);
        }

        svg.push(`
          <text
            x="${x}"
            y="164"
            text-anchor="middle"
            class="h3c-chart-label"
          >
            ${
              count === 12
                ? item.month
                : `${item.month}月`
            }
          </text>
        `);
      }
    );

    return `
      <details
        class="h3c-trend-details"
        ${state.trendOpen ? "open" : ""}
      >
        <summary>
          最近 ${count} 个月工作日工时趋势
        </summary>

        <div class="h3c-trend-description">
          趋势不包含周末加班，单位为小时。
          圆点表示暂无缓存。
        </div>

        <svg
          class="h3c-trend-chart"
          viewBox="0 0 ${width} ${height}"
          role="img"
          aria-label="最近工作日工时趋势图"
        >
          ${svg.join("")}
        </svg>
      </details>
    `;
  }

  function clearColors(cells) {

  function applyColors(
    result,
    summary
  ) {
    clearColors(result.cells);

    if (
      !state.settings
        .calendarColorEnabled
    ) {
      return;
    }

    for (
      const detail
      of summary.details
    ) {
      const cell =
        result.cellsByDate.get(
          detail.dateKey
        );

      if (!cell) {
        continue;
      }

      let status = "zero";
      let tip = "工时持平";

      if (
        detail.isSpecial &&
        detail.leaveStart != null
      ) {
        if (detail.isInvalid) {
          status = "invalid";
          tip =
            `打卡异常：` +
            detail.invalidReasons.join("；");
        } else if (detail.overtimeMinutes > 0) {
          status = "positive";
          tip =
            `工时盈余 ` +
            formatHours(
              detail.overtimeMinutes
            );
        } else if (
          detail.overtimeMinutes < 0
        ) {
          status = "negative";
          tip =
            `工时缺少 ` +
            formatHours(
              detail.overtimeMinutes,
              false
            );
        } else {
          status = "zero";
          tip = "工时持平";
        }
      }
      else if (detail.isSpecial) {
        status = "special";
        tip =
          `${detail.dayLabel}：` +
          "工时豁免";
      } else if (detail.isInvalid) {
        status = "invalid";
        tip =
          `打卡异常：` +
          detail.invalidReasons.join("；");
      } else if (
        detail.isWeekend &&
        detail.overtimeMinutes > 0
      ) {
        status = "weekend";
        tip =
          `周末加班 ` +
          formatHours(
            detail.overtimeMinutes
          );
      } else if (
        detail.overtimeMinutes > 0
      ) {
        status = "positive";
        tip =
          `工时盈余 ` +
          formatHours(
            detail.overtimeMinutes
          );
      } else if (
        detail.overtimeMinutes < 0
      ) {
        status = "negative";
        tip =
          `工时缺少 ` +
          formatHours(
            detail.overtimeMinutes,
            false
          );
      }

      cell.dataset.h3cStatus =
        status;

      cell.dataset.h3cTip =
        tip;
    }
  }

  function renderSuggestion(
    summary,
    remaining
  ) {
    if (!remaining.current) {
      return `
        <div class="h3c-suggestion">
          <strong>
            ${
              remaining.future
                ? "未来月份"
                : "历史月份"
            }
          </strong>

          <span>
            ${
              remaining.future
                ? `预计有 ${remaining.count} 个工作日。`
                : "历史月份不计算剩余工作日。"
            }
          </span>
        </div>
      `;
    }

    if (remaining.count <= 0) {
      return `
        <div class="h3c-suggestion">
          <strong>补时建议</strong>

          <span>
            本月已没有剩余工作日。
          </span>
        </div>
      `;
    }

    if (summary.totalMinutes < 0) {
      const deficit =
        Math.abs(
          summary.totalMinutes
        );

      const daily =
        Math.ceil(
          deficit /
          remaining.count
        );

      return `
        <div
          class="
            h3c-suggestion
            h3c-suggestion-warning
          "
        >
          <strong>
            剩余工作日补时建议
          </strong>

          <span>
            还有 ${remaining.count} 个工作日，
            当前工作日缺时

            <b class="h3c-negative">
              ${escapeHtml(
                formatShort(deficit)
              )}
            </b>，

            平均每天建议补

            <b>
              ${escapeHtml(
                formatShort(daily)
              )}
            </b>。
          </span>
        </div>
      `;
    }

    const margin =
      Math.floor(
        summary.totalMinutes /
        remaining.count
      );

    return `
      <div
        class="
          h3c-suggestion
          h3c-suggestion-success
        "
      >
        <strong>
          ${
            summary.totalMinutes > 0
              ? "工作日工时有盈余"
              : "工作日工时正好持平"
          }
        </strong>

        <span>
          还有 ${remaining.count} 个工作日。

          ${
            summary.totalMinutes > 0
              ? (
                `平均每天余量约 ` +
                `${escapeHtml(
                  formatShort(margin)
                )}。`
              )
              : "后续完成标准工时即可。"
          }
        </span>
      </div>
    `;
  }

    if (
      detail.clockIn === null &&
      detail.clockOut === null
    ) {
      return detail.isSpecial
        ? "特殊考勤，无需打卡"
        : "无";
    }

    return (
      `${
        detail.clockIn === null
          ? "无"
          : formatClock(
              detail.clockIn
            )
      } - ${
        detail.clockOut === null
          ? "无"
          : formatClock(
              detail.clockOut
            )
      }`
    );
  }

  function rangeText(year, month) {
    const today =
      new Date();

    if (
      today.getFullYear() === year &&
      today.getMonth() + 1 === month
    ) {
      const yesterday =
        new Date(
          today.getFullYear(),
          today.getMonth(),
          today.getDate() - 1
        );

      if (
        yesterday.getFullYear() !== year ||
        yesterday.getMonth() + 1 !== month
      ) {
        return (
          `${year}年${month}月` +
          "暂无可统计日期"
        );
      }

      return (
        `${year}年${month}月1日 - ` +
        `${month}月` +
        `${yesterday.getDate()}日（昨天）`
      );
    }

    const lastDay =
      new Date(
        year,
        month,
        0
      ).getDate();

    return (
      `${year}年${month}月1日 - ` +
      `${month}月${lastDay}日`
    );
  }

  function readPageStatistics() {
    const statText =
      Array.from(
        document.body.querySelectorAll(
          ".place-self-center-item-iconitem"
        )
      )
        .map(element =>
          (
            element.textContent || ""
          ).replace(/\s+/g, "")
        )
        .join(" ");

    const forgotMatch =
      statText.match(
        /忘刷卡(\d+)次/
      );

    state.forgotPunchCount =
      forgotMatch
        ? Number(forgotMatch[1])
        : null;

    const annualMatch =
      statText.match(
        /预配年假余额([0-9.]+)天/
      );

    const balance = annualMatch
      ? Number(annualMatch[1])
      : null;

    state.annualLeaveBalance =
      balance !== null &&
      Number.isFinite(balance)
        ? balance
        : null;
  }

  function forgotPunchClass() {
    if (
      state.forgotPunchCount === null
    ) {
      return "h3c-zero";
    }

    return state.forgotPunchCount >= 3
      ? "h3c-warning"
      : "h3c-positive";
  }

  function forgotPunchText() {
    if (
      state.forgotPunchCount === null
    ) {
      return "--";
    }

    return (
      `${state.forgotPunchCount} / ` +
      `4 次`
    );
  }

  function annualLeaveClass() {
    if (
      state.annualLeaveBalance === null
    ) {
      return "h3c-zero";
    }

    return state.annualLeaveBalance < 1
      ? "h3c-danger"
      : "h3c-zero";
  }

      state.annualLeaveBalance === null
    ) {
      return "--";
    }

    return (
      `${state.annualLeaveBalance.toFixed(2)} 天`
    );
  }

  function getFormSettings() {
    const values = {};

    for (const key of TIME_KEYS) {
      values[key] =
        state.panel.querySelector(
          `[data-setting="${key}"]`
        ).value;
    }

    values.theme =
      state.panel.querySelector(
        '[data-select="theme"]'
      ).value;

    values.trendMonths =
      Number(
        state.panel.querySelector(
          '[data-select="trendMonths"]'
        ).value
      );

    values.calendarColorEnabled =
      state.panel.querySelector(
        '[data-checkbox="calendarColorEnabled"]'
      ).checked;

    values.panelWidth =
      Number(
        state.panel.querySelector(

  function validateSettings(values) {
    const parsed = {};

    for (const key of TIME_KEYS) {
      parsed[key] =
        timeToMinutes(
          values[key]
        );
    }

    if (
      Object.values(parsed).some(
        value =>
          value === null
      )
    ) {
      return {
        valid: false,
        message:
          "请完整填写所有时间。"
      };
    }

    if (
      parsed.normalEnd <=
      parsed.normalStart
    ) {
      return {
        valid: false,
        message:
          "正常下班时间必须晚于上班时间。"
      };
    }

    if (
      parsed.flexibleLatest <
      parsed.normalStart
    ) {
      return {
        valid: false,
        message:
          "弹性最晚时间不能早于正常上班时间。"
      };
    }

    if (
      parsed.lunchEnd <=
      parsed.lunchStart
    ) {
      return {
        valid: false,
        message:
          "午休结束时间必须晚于开始时间。"
      };
    }

    if (
      standardMinutes(parsed) <= 0
    ) {
      return {
        valid: false,
        message:
          "标准净工时必须大于0。"
      };
    }

    if (
      values.panelWidth < 380 ||
      values.panelWidth > 700
    ) {
      return {
        valid: false,
        message:
          "完整模式宽度必须在380到700像素之间。"
      };
    }

    if (
      values.compactWidth < 280 ||
      values.compactWidth > 450
    ) {
      return {
        valid: false,
        message:
          "紧凑模式宽度必须在280到450像素之间。"
      };
    }

    if (
      values.panelOpacity < 60 ||
      values.panelOpacity > 100
    ) {
      return {
        valid: false,
        message:
          "面板透明度必须在60%到100%之间。"
      };
    }

    return {
      valid: true,
      parsed
    };
  }

  function updateRangeOutputs() {
    if (!state.panel) {
      return;
    }

    state.panel
      .querySelectorAll(
        "[data-range]"
      )
      .forEach(input => {
        const key =
          input.dataset.range;

        const output =
          state.panel.querySelector(
            `[data-output="${key}"]`
          );

        if (!output) {
          return;
        }

        output.textContent =
          `${input.value}` +
          `${input.dataset.unit || ""}`;
      });
  }

  function applyAppearance(
    settings = state.settings
  ) {
    if (!state.panel) {
      return;
    }

    const panelWidth =
      clamp(
        settings.panelWidth,
        380,
        700,
        DEFAULT_SETTINGS.panelWidth
      );

    const compactWidth =
      clamp(
        settings.compactWidth,
        280,
        450,
        DEFAULT_SETTINGS.compactWidth
      );

    const opacity =
      clamp(
        settings.panelOpacity,
        60,
        100,
        DEFAULT_SETTINGS.panelOpacity
      );

    state.panel.style.setProperty(
      "--h3c-panel-width",
      `${panelWidth}px`
    );

    state.panel.style.setProperty(
      "--h3c-compact-width",
      `${compactWidth}px`
    );

    state.panel.style.setProperty(
      "--h3c-panel-opacity",
      String(opacity / 100)
    );
  }

  function applyTheme(
    theme = state.settings.theme
  ) {
    if (state.panel) {
      state.panel.dataset.theme =
        theme;
    }
  }

  function updateSettingsPreview() {
    const values =
      getFormSettings();

    const validation =
      validateSettings(values);

    const preview =
      state.panel.querySelector(
        ".h3c-setting-preview"
      );

    updateRangeOutputs();

    if (!validation.valid) {
      preview.textContent =
        validation.message;

      return;
    }

    const themeLabel =
      THEMES.find(
        ([value]) =>
          value === values.theme
      )?.[1] || "未知";

    preview.innerHTML = `
      当前主题：
      ${escapeHtml(themeLabel)}
      <br>

      面板尺寸：
      ${values.panelWidth}px /
      ${values.compactWidth}px
      <br>

      面板透明度：
      ${values.panelOpacity}%
      <br>

      日历着色：
      ${
        values.calendarColorEnabled
          ? "开启"
          : "关闭"
      }
      <br>

      标准净工时：
      ${escapeHtml(
        formatShort(
          standardMinutes(
            validation.parsed
          )
        )
      )}
    `;
  }

      return;

      state.settings.theme
    );

    applyAppearance(
      state.settings
    );

    state.panel
      .querySelector(
        ".h3c-settings-mask"
      )
      ?.classList.remove(
        "h3c-show"
      );

    state.panel.classList.remove(
      "h3c-settings-open"
    );

    requestAnimationFrame(
      keepInsideViewport
    );
  }

  function saveSettings() {
    const values =
      getFormSettings();

    const validation =
      validateSettings(values);

    const errorElement =
      state.panel.querySelector(
        ".h3c-setting-error"
      );

    if (!validation.valid) {
      errorElement.textContent =
        validation.message;

      return;
    }

    errorElement.textContent = "";

    state.settings = {
      ...values,

      trendMonths:
        Number(values.trendMonths),

      panelWidth:
        Number(values.panelWidth),

      compactWidth:
        Number(values.compactWidth),

      panelOpacity:
        Number(values.panelOpacity)
    };

    writeStorage(
      CONFIG.settingsKey,
      state.settings
    );

    applyTheme(
      state.settings.theme
    );

    applyAppearance(
      state.settings
    );

    updateColorButtons();
    closeSettings();

    state.cellCache.clear();
    state.lastSignature = "";
    state.colorsApplied = false;

    scheduleRefresh(0);
  }

  function bindPanelEvents() {
    state.panel.addEventListener(
      "click",
      event => {
        const actionElement =
          event.target instanceof Element
            ? event.target.closest(
                "[data-action]"
              )
            : null;

        if (!actionElement) {
          return;
        }

        event.stopPropagation();

        const action =
          actionElement.dataset.action;

        switch (action) {
          case "refresh":
            state.cellCache.clear();
            state.lastSignature = "";
            state.colorsApplied = false;
            state.colorsSignature = "";

            actionElement.classList.add(
              "h3c-refreshing"
            );

            window.setTimeout(
              () => {
                actionElement.classList.remove(
                  "h3c-refreshing"
                );
              },
              500
            );

            scheduleRefresh(0);
            break;

          case "toggle-colors":
            toggleCalendarColors();
            break;

          case "settings":
            openSettings();
            break;

          case "compact":
            setDisplayMode("compact");
            break;

          case "expand":
            setDisplayMode("normal");
            break;

          case "minimize":
            setDisplayMode("mini");
            break;

          case "cancel-settings":
            closeSettings();
            break;

          case "save-settings":
            saveSettings();
            break;

          case "reset-settings":
            resetSettingsForm();
            break;

          case "clear-history":
            if (
              window.confirm(
                "确定清除本地趋势数据吗？"
              )
            ) {
              state.history = {};

              deleteStorage(
                CONFIG.historyKey
              );

              clearTimeout(
                state.stableTimer
              );

              state.stableTimer = null;
              state.pendingHistoryToken = "";
              state.lastSignature = "";

              closeSettings();
              scheduleRefresh(0);
            }
            break;
        }
      }
    );

    state.panel
      .querySelectorAll(
        ".h3c-time-input, .h3c-select"
      )
      .forEach(element => {
        const update = () => {
          if (
            element.dataset.select ===
            "theme"
          ) {
            applyTheme(
              element.value
            );
          }

          updateSettingsPreview();
        };

        element.addEventListener(
          "input",
          update
        );

        element.addEventListener(
          "change",
          update
        );
      });

    state.panel
      .querySelectorAll(
        ".h3c-range-input"
      )
      .forEach(element => {
        element.addEventListener(
          "input",
          () => {
            const values =
              getFormSettings();

            applyAppearance(values);
            updateSettingsPreview();
          }
        );
      });

    state.panel
      .querySelector(
        '[data-checkbox="calendarColorEnabled"]'
      )
      .addEventListener(
        "change",
        updateSettingsPreview
      );

    const mask =
      state.panel.querySelector(
        ".h3c-settings-mask"
      );

    mask.addEventListener(
      "pointerdown",
      event => {
        if (event.target === mask) {
          closeSettings();
        }
      }
    );

    mask.addEventListener(
      "wheel",
      event => {
        if (state.settingsOpen) {
          event.stopPropagation();
        }
      },
      {
        passive: true
      }
    );

    const mini =
      state.panel.querySelector(
        ".h3c-mini-view"
      );

    mini.addEventListener(
      "keydown",
      event => {
        if (
          event.key === "Enter" ||
          event.key === " "
        ) {
          event.preventDefault();

          setDisplayMode(
            "compact"
          );
        }
      }
    );

    document.addEventListener(
      "keydown",
      event => {
        if (
          event.key === "Escape" &&
          state.settingsOpen
        ) {
          event.preventDefault();
          closeSettings();
        }
      }
    );
  }

  function setDisplayMode(mode) {
    if (
      !DISPLAY_MODES.includes(mode)
    ) {

      CONFIG.displayModeKey,
      mode
    );

    applyDisplayMode();

    requestAnimationFrame(
      () => {
        keepInsideViewport();
        savePosition();
      }
    );
  }

  function applyDisplayMode() {
    if (state.panel) {
      state.panel.dataset.displayMode =
        state.displayMode;
    }
  }

  function renderSignature(result) {
    return JSON.stringify({
      month:
        result.displayedMonth,

      settings:
        state.settings,

      records:
        result.records.map(
          record => [
            record.dateKey,
            record.clockIn,
            record.clockOut,
            record.dayType
          ]
        )
    });
  }

  function refreshNow() {
    if (!isTargetPage()) {
      return;
    }

    const panel =
      createPanel();

    const content =
      panel.querySelector(
        ".h3c-content"
      );

    try {
      attachCalendarObserver();

      readPageStatistics();

      const result =
        extractRecords();

      if (!result.displayedMonth) {
        state.latestResult = null;
        state.latestSummary = null;
        state.latestMonth = null;
        state.lastSignature = "";

        updateModeViews();

        content.innerHTML = `
          <div class="h3c-empty">
            暂未找到考勤日历，
            请等待页面加载完成。
          </div>
        `;

        return;
      }

      const signature =
        renderSignature(result);

      if (
        signature ===
          state.lastSignature &&
        state.latestSummary
      ) {
        state.latestResult =
          result;

        // 性能：日历格子着色状态由 cell 的
        // dataset 承载，signature 未变时
        // 着色结果必然相同，跳过重刷
        if (
          !state.colorsApplied ||
          state.colorsSignature !== signature
        ) {
          applyColors(
            result,
            state.latestSummary
          );

          state.colorsSignature =
            signature;

          state.colorsApplied = true;
        }

        return;
      }

      const summary =
        summarize(
          result.records
        );

      const remaining =
        remainingWorkdays(
          result.displayedMonth,
          result.cells
        );

      state.lastSignature =
        signature;

      state.latestResult =
        result;

      state.latestSummary =
        summary;

      state.latestMonth =
        result.displayedMonth;

      stageHistorySave(result);

      applyColors(
        result,
        summary
      );

      state.colorsSignature =
        signature;

      state.colorsApplied = true;

      updateModeViews();
      updateColorButtons();

      content.innerHTML =
        renderFull(
          result,
          summary,
          remaining
        );

      bindDynamicEvents(content);
    } catch (error) {
      console.error(
        "[H3C 考勤助手] 计算失败：",
        error
      );

      state.lastSignature = "";

      content.innerHTML = `
        <div class="h3c-empty">
          计算失败：
          ${escapeHtml(
            error.message
          )}
        </div>
      `;
    }
  }

    );
  }

  function cancelIdleTask() {
    if (state.idleId === null) {
      return;
    }

    if (
      typeof cancelIdleCallback ===
      "function"
    ) {
      cancelIdleCallback(
        state.idleId
      );
    } else {
      clearTimeout(
        state.idleId
      );
    }

    state.idleId = null;
  }

  function scheduleRefresh(
    delay = CONFIG.refreshDelay
  ) {
    clearTimeout(
      state.refreshTimer
    );

    cancelIdleTask();

    state.refreshTimer =
      window.setTimeout(
        () => {
          const run = () => {
            state.idleId = null;
            refreshNow();
          };

          if (
            typeof requestIdleCallback ===
            "function"
          ) {
            state.idleId =
              requestIdleCallback(
                run,
                {
                  timeout: 700
                }
              );
          } else {
            state.idleId =
              window.setTimeout(
                run,
                0
              );
          }
        },
        delay
      );
  }

  function detachCalendarObserver() {
    if (
      state.calendar &&
      state.calendarClickHandler
    ) {
      state.calendar.removeEventListener(
        "click",
        state.calendarClickHandler,
        true
      );
    }

    state.calendarObserver
      ?.disconnect();

    state.calendarObserver = null;
    state.calendarClickHandler = null;
    state.calendar = null;
  }

  function attachCalendarObserver() {
    const calendar =
      findCalendar();

    if (!calendar) {
      return;
    }

    if (
      calendar === state.calendar &&
      state.calendarObserver
    ) {
      return;
    }

    detachCalendarObserver();

    state.calendar =
      calendar;

    state.calendarObserver =
      new MutationObserver(
        mutations => {
          const relevant =
            mutations.some(
              mutation => {
                if (
                  mutation.type ===
                    "childList" ||
                  mutation.type ===
                    "characterData"
                ) {
                  return true;
                }

                return (
                  mutation.type ===
                    "attributes" &&
                  [
                    "title",
                    "class"
                  ].includes(
                    mutation.attributeName
                  )
                );
              }
            );

          if (relevant) {
            scheduleRefresh();
          }
        }
      );

    state.calendarObserver.observe(
      calendar,
      {
        childList: true,
        subtree: true,
        characterData: true,
        attributes: true,

        attributeFilter: [
          "title",
          "class"
        ]
      }
    );

    state.calendarClickHandler =
      () => {
        window.setTimeout(
          () => {
            state.cellCache.clear();
            state.lastSignature = "";
            state.colorsApplied = false;
            state.colorsSignature = "";

            scheduleRefresh(0);
          },
          120
        );
      };

    calendar.addEventListener(
      "click",
      state.calendarClickHandler,
      true
    );
  }

  function installDiscoveryObserver() {
    if (
      state.discoveryObserver ||
      !document.body
    ) {
      return;
    }

    state.discoveryObserver =
      new MutationObserver(
        () => {
          if (!isTargetPage()) {
            return;
          }

          if (
            state.calendar?.isConnected
          ) {
            return;
          }

          detachCalendarObserver();
          attachCalendarObserver();
          scheduleRefresh();
        }
      );

    state.discoveryObserver.observe(
      document.body,
      {
        childList: true,
        subtree: true
      }
    );
  }

  function bindDrag(
    handle,
    ignoreButtons,
    clickAction
  ) {
    if (!handle) {
      return;
    }

    let active = false;
    let moved = false;
    let pointerId = null;

    let startX = 0;
    let startY = 0;
    let startLeft = 0;
    let startTop = 0;

    let panelWidth = 0;
    let panelHeight = 0;

    let deltaX = 0;
    let deltaY = 0;

    let frameId = null;

    function boundedDelta() {
      const maxLeft =
        Math.max(
          0,
          window.innerWidth -
            panelWidth
        );

      const maxTop =
        Math.max(
          0,
          window.innerHeight -
            Math.min(
              panelHeight,
              60
            )
        );

      const left =
        Math.min(
          maxLeft,
          Math.max(
            0,
            startLeft + deltaX
          )
        );

      const top =
        Math.min(
          maxTop,
          Math.max(
            0,
            startTop + deltaY
          )
        );

      return {
        x:
          left - startLeft,

        y:
          top - startTop
      };
    }

    function renderFrame() {
      frameId = null;

      if (!active) {
        return;
      }

      const delta =
        boundedDelta();

      state.panel.style.transform =
        `translate3d(` +
        `${delta.x}px,` +
        `${delta.y}px,0)`;
    }

    function onMove(event) {
      if (
        !active ||
        event.pointerId !== pointerId
      ) {
        return;
      }

      deltaX =
        event.clientX - startX;

      deltaY =
        event.clientY - startY;

      if (
        !moved &&
        Math.hypot(
          deltaX,
          deltaY
        ) >= CONFIG.dragThreshold
      ) {
        moved = true;
      }

      if (
        moved &&
        frameId === null
      ) {
        frameId =
          requestAnimationFrame(
            renderFrame
          );
      }

      event.preventDefault();
    }

    function finish(
      event,
      cancelled
    ) {
      if (
        !active ||
        (
          event &&
          event.pointerId !== pointerId
        )
      ) {
        return;
      }

      const wasMoved =
        moved;

      active = false;

      if (frameId !== null) {
        cancelAnimationFrame(
          frameId
        );

        frameId = null;
      }

      if (
        wasMoved &&
        !cancelled
      ) {
        const delta =
          boundedDelta();

        state.panel.style.left =
          `${startLeft + delta.x}px`;

        state.panel.style.top =
          `${startTop + delta.y}px`;

        state.panel.style.right =
          "auto";
      }

      state.panel.style.transform =
        "";

      state.panel.classList.remove(
        "h3c-dragging"
      );

      handle.classList.remove(
        "h3c-handle-dragging"
      );

      state.dragging = false;

      window.removeEventListener(
        "pointermove",
        onMove,
        true
      );

      window.removeEventListener(
        "pointerup",
        onUp,
        true
      );

      window.removeEventListener(
        "pointercancel",
        onCancel,
        true
      );

      if (
        wasMoved &&
        !cancelled
      ) {
        savePosition();
      } else if (
        !wasMoved &&
        !cancelled &&
        clickAction
      ) {
        clickAction();
      }

      moved = false;
      pointerId = null;
    }

    function onUp(event) {
      finish(event, false);
    }

    function onCancel(event) {
      finish(event, true);
    }

    handle.addEventListener(
      "pointerdown",
      event => {
        if (
          state.settingsOpen ||
          (
            event.pointerType ===
              "mouse" &&
            event.button !== 0
          )
        ) {
          return;
        }

        if (
          ignoreButtons &&
          event.target instanceof Element &&
          event.target.closest("button")
        ) {
          return;
        }

        const rect =
          state.panel
            .getBoundingClientRect();

        active = true;
        moved = false;

        pointerId =
          event.pointerId;

        startX =
          event.clientX;

        startY =
          event.clientY;

        startLeft =
          rect.left;

        startTop =
          rect.top;

        panelWidth =
          rect.width;

        panelHeight =
          rect.height;

        deltaX = 0;
        deltaY = 0;

        state.panel.style.left =
          `${startLeft}px`;

        state.panel.style.top =
          `${startTop}px`;

        state.panel.style.right =
          "auto";

        state.dragging = true;

        state.panel.classList.add(
          "h3c-dragging"
        );

        handle.classList.add(
          "h3c-handle-dragging"
        );

        window.addEventListener(
          "pointermove",
          onMove,
          {
            capture: true,
            passive: false
          }
        );

        window.addEventListener(
          "pointerup",
          onUp,
          true
        );

        window.addEventListener(
          "pointercancel",
          onCancel,
          true
        );

        event.preventDefault();
      }
    );

    if (clickAction) {
      handle.addEventListener(
        "click",
        event => {
          event.preventDefault();
          event.stopPropagation();
        }
      );
    }
  }

  // 面板位置
  // ============================================================

  function savePosition() {
    if (
      !state.panel?.isConnected
    ) {
      return;
    }

    const rect =
      state.panel
        .getBoundingClientRect();

    writeStorage(
      CONFIG.positionKey,
      {
        left: rect.left,
        top: rect.top
      }
    );
  }

  function restorePosition() {
    const position =
      readStorage(
        CONFIG.positionKey,
        null
      );

    if (
      !position ||
      !Number.isFinite(
        position.left
      ) ||
      !Number.isFinite(
        position.top
      )
    ) {
      return;
    }

    state.panel.style.left =
      `${Math.max(
        0,
        position.left
      )}px`;

    state.panel.style.top =
      `${Math.max(
        0,
        position.top
      )}px`;

    state.panel.style.right =
      "auto";

    requestAnimationFrame(
      keepInsideViewport
    );
  }

  function keepInsideViewport() {
    if (
      !state.panel?.isConnected ||
      state.dragging ||
      state.settingsOpen
    ) {
      return;
    }

    const rect =
      state.panel
        .getBoundingClientRect();

    const maxLeft =
      Math.max(
        0,
        window.innerWidth -
          state.panel.offsetWidth
      );

    const maxTop =
      Math.max(
        0,
        window.innerHeight -
          Math.min(
            state.panel.offsetHeight,
            60
          )
      );

    state.panel.style.left =
      `${Math.max(
        0,
        Math.min(
          rect.left,
          maxLeft
        )
      )}px`;

    state.panel.style.top =
      `${Math.max(
        0,
        Math.min(
          rect.top,
          maxTop
        )
      )}px`;

    state.panel.style.right =
      "auto";
  }
